cmake_minimum_required(VERSION 3.16)
project(LegacyUiTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Widgets Sql Network Test)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Widgets Sql Network Test)
if(NOT TARGET ElaWidgetTools)
    add_subdirectory(../../third_party/ElaWidgetTools ela)
endif()
file(GLOB_RECURSE LEGACY_APP_SOURCES CONFIGURE_DEPENDS ../../src/*.cpp ../../src/*.h)
list(FILTER LEGACY_APP_SOURCES EXCLUDE REGEX "/main.cpp$")
configure_file(../../src/AppVersion.h.in generated/AppVersion.h @ONLY)
add_executable(legacy_ui_tests LegacyUiTests.cpp ${LEGACY_APP_SOURCES} ../../resources/resources.qrc)
set_target_properties(legacy_ui_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_include_directories(legacy_ui_tests PRIVATE ../../src ../../src/Types ../../src/Utils ../../src/Cal
    ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(legacy_ui_tests PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Widgets
    Qt${QT_VERSION_MAJOR}::Sql Qt${QT_VERSION_MAJOR}::Network Qt${QT_VERSION_MAJOR}::Test ElaWidgetTools)
if(MSVC)
    target_compile_options(legacy_ui_tests PRIVATE /utf-8)
endif()
add_custom_command(TARGET legacy_ui_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:ElaWidgetTools> $<TARGET_FILE_DIR:legacy_ui_tests>)
enable_testing()
add_test(NAME legacy_ui_features COMMAND legacy_ui_tests -platform windows -o -,txt)
set_tests_properties(legacy_ui_features PROPERTIES TIMEOUT 45)
