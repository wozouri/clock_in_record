cmake_minimum_required(VERSION 3.16)
project(StorageCompatibilityTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Sql)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Sql)
add_executable(storage_compatibility_tests main.cpp ../../src/Data/AttendanceStorage.cpp)
set_target_properties(storage_compatibility_tests PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_include_directories(storage_compatibility_tests PRIVATE ../../src)
target_link_libraries(storage_compatibility_tests PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Sql)
if(MSVC)
    target_compile_options(storage_compatibility_tests PRIVATE /utf-8)
endif()
enable_testing()
foreach(version 4 5 6 7)
    add_test(NAME storage_schema_${version} COMMAND storage_compatibility_tests ${version})
endforeach()
